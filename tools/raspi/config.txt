# zCore Raspberry Pi 4 / Pi 400 boot configuration
#
# This file is read by the Pi firmware (start4.elf) to configure
# the hardware before loading the kernel.

# Boot in 64-bit mode (AArch64)
arm_64bit=1

# Use the PL011 UART (uart0) as the primary serial port.
# The Pi 4 has two UARTs:
#   - PL011 (uart0) at 0xFE201000 -- full-featured, stable baud rate
#   - Mini-UART (uart1) at 0xFE215040 -- baud tied to core clock
# We want PL011 for reliable serial output.
enable_uart=1
dtoverlay=disable-bt

# Disable the GPU splash screen (rainbow square)
disable_splash=1

# Set core clock to fixed frequency (required for stable mini-UART baud
# if it were used; good practice regardless)
core_freq=500
core_freq_min=500

# Set the kernel filename (default is kernel8.img for AArch64)
kernel=kernel8.img

# Serial console baud rate
init_uart_baud=115200

# Enable the GIC (Generic Interrupt Controller).
# NOTE: Timer interrupts are not yet working due to GIC Group 0/1
# configuration issue -- see GitHub issue for details.
enable_gic=1

# Load the rootfs initrd image into RAM after the kernel.
# The firmware sets linux,initrd-start/end in the DTB so
# the kernel can find and mount it.
initramfs initrd.img followkernel
